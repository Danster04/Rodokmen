/** * Zobrazení detailu konkrétní osoby v bočním panelu */ function showPersonDetails(person) { AppState.selectedPersonId = person.id; const detailPanel = document.getElementById('detail-panel'); if (!detailPanel) return; // Základní informace o osobě let html = `

Detail osoby

JMÉNO ${person.name}
NAROZENÍ ${person.birth || 'Neuvedeno'}
STAV / VZTAH ${person.status || 'Neuvedeno'}
`; // Sekce dětí – zde je natvrdo vynucen pouze nápis "DĚTI", ignorují se sourozenci i matka if (person.children && person.children.length > 0) { html += `
`; html += `DĚTI`; html += `
    `; person.children.forEach(child => { html += `
  • ${child.name}
    • `; }); html += `
`; } // Úmrtí a ID v databázi html += `
ÚMRTÍ ${person.death || 'Neuvedeno'}
ID V DATABÁZI ${person.dbId || ''}
`; detailPanel.innerHTML = html; }

function processGedcomData(text) { const lines = text.split(/\r?\n/); let individuals = {}; let families = {}; let sharedEvents = {}; let mediaFiles = {}; let currentId = null; let currentType = null; let currentEvent = null; let currentFamilyId = null; let currentEventId = null; let currentNameObj = null; lines.forEach(line => { const match = line.match(/^([0-9])\s+(@\w+@|[A-Z_]+)\s*(.*)$/); if (!match) return; const level = parseInt(match[1]); const key = match[2]; const val = match[3]; if (level === 0) { currentEvent = null; currentNameObj = null; currentEventId = null; if (key.startsWith('@') && key.endsWith('@')) { if (val === 'INDI') { currentId = key; individuals[currentId] = { id: currentId, firstName: '', marriedName: '', birthName: '', sex: 'U', birthDate: '', birthPlace: '', deathDate: '', deathPlace: '', occupation: '', marriageDate: '', marriagePlace: '', hasSingleFamily: false, parents: [], children: [], spouses: [], siblings: [], media: [] }; currentFamilyId = null; currentType = 'INDI'; } else if (val === 'FAM') { currentFamilyId = key; families[currentFamilyId] = { id: currentFamilyId, husb: null, wife: null, children: [], marriageDate: '', marriagePlace: '', sharedEventRef: null }; currentId = null; currentType = 'FAM'; } else if (val === 'EVEN') { currentEventId = key; sharedEvents[currentEventId] = { id: currentEventId, type: '', date: '', place: '' }; currentType = 'EVEN'; } else if (val === 'OBJE') { currentId = key; mediaFiles[currentId] = { id: currentId, file: '' }; currentType = 'OBJE'; } else { currentId = null; currentFamilyId = null; currentType = null; } } } else if (level === 1) { currentEvent = null; if (currentType === 'INDI' && currentId && individuals[currentId]) { if (key === 'NAME') { currentNameObj = { fullName: val.replace(/\//g, '').trim(), type: '' }; if (!individuals[currentId].allNames) individuals[currentId].allNames = []; individuals[currentId].allNames.push(currentNameObj); } else if (key === 'SEX') { individuals[currentId].sex = val.trim(); } else if (key === 'BIRT' || key === 'DEAT' || key === 'OCCU' || key === 'MARR' || key === 'EVEN') { currentEvent = key; if (key === 'OCCU') { individuals[currentId].occupation = val.trim(); } } else if (key === 'OBJE') { individuals[currentId].media.push(val.trim()); } } else if (currentType === 'FAM' && currentFamilyId && families[currentFamilyId]) { if (key === 'HUSB') families[currentFamilyId].husb = val.trim(); if (key === 'WIFE') families[currentFamilyId].wife = val.trim(); if (key === 'CHIL') families[currentFamilyId].children.push(val.trim()); if (key === 'MARR') { currentEvent = 'MARR'; } else if (key === 'EVEN') { families[currentFamilyId].sharedEventRef = val.replace(/@/g, '').trim(); } } else if (currentType === 'EVEN' && currentEventId && sharedEvents[currentEventId]) { if (key === 'TYPE') sharedEvents[currentEventId].type = val.trim().toLowerCase(); } else if (currentType === 'OBJE' && currentId && mediaFiles[currentId]) { if (key === 'FILE') mediaFiles[currentId].file = val.replace(/["']/g, '').trim(); } } else if (level === 2) { if (currentType === 'INDI' && currentId && individuals[currentId]) { if (currentNameObj && key === 'TYPE') { currentNameObj.type = val.trim().toLowerCase(); } if (currentEvent === 'BIRT') { if (key === 'DATE') individuals[currentId].birthDate = formatGedcomDate(val.trim()); if (key === 'PLAC') individuals[currentId].birthPlace = val.trim(); } if (currentEvent === 'DEAT') { if (key === 'DATE') individuals[currentId].deathDate = formatGedcomDate(val.trim()); if (key === 'PLAC') individuals[currentId].deathPlace = val.trim(); } if (currentEvent === 'MARR') { if (key === 'DATE') individuals[currentId].marriageDate = formatGedcomDate(val.trim()); if (key === 'PLAC') individuals[currentId].marriagePlace = val.trim(); } if (currentEvent === 'EVEN') { if (key === 'TYPE' && val.trim().toLowerCase() === 'marriage') { individuals[currentId].isMarriageEvent = true; } if (individuals[currentId].isMarriageEvent) { if (key === 'DATE') individuals[currentId].marriageDate = formatGedcomDate(val.trim()); if (key === 'PLAC') individuals[currentId].marriagePlace = val.trim(); } } } else if (currentType === 'FAM' && currentFamilyId && families[currentFamilyId]) { if (currentEvent === 'MARR') { if (key === 'DATE') families[currentFamilyId].marriageDate = formatGedcomDate(val.trim()); if (key === 'PLAC') families[currentFamilyId].marriagePlace = val.trim(); } } else if (currentType === 'EVEN' && currentEventId && sharedEvents[currentEventId]) { if (key === 'DATE') sharedEvents[currentEventId].date = formatGedcomDate(val.trim()); if (key === 'PLAC') sharedEvents[currentEventId].place = val.trim(); } } }); Object.values(individuals).forEach(p => { if (p.allNames && p.allNames.length > 0) { let married = p.allNames.find(n => n.type.includes('manžel') || n.type.includes('married')); let birth = p.allNames.find(n => n.type.includes('rodn') || n.type.includes('birth')); if (!married && p.allNames.length > 0) married = p.allNames[0]; if (!birth && p.allNames.length > 1) birth = p.allNames[1]; p.marriedName = married ? married.fullName : p.allNames[0].fullName; p.birthName = birth ? birth.fullName : ''; } else { p.marriedName = 'Neznámý'; } p.displayName = p.marriedName; if (p.birthName && p.birthName !== p.marriedName) { p.detailName = `${p.marriedName} (roz. ${p.birthName})`; } else { p.detailName = p.marriedName; } }); Object.values(families).forEach(fam => { let husb = fam.husb && individuals[fam.husb] ? fam.husb : null; let wife = fam.wife && individuals[fam.wife] ? fam.wife : null; let kids = fam.children.filter(c => c && individuals[c]); if ((husb && !wife) || (!husb && wife)) { let singleParentId = husb || wife; if (kids.length > 0) { individuals[singleParentId].hasSingleFamily = true; } } let mDate = fam.marriageDate; let mPlace = fam.marriagePlace; if (!mDate && husb && individuals[husb].marriageDate) { mDate = individuals[husb].marriageDate; mPlace = individuals[husb].marriagePlace; } if (!mDate && wife && individuals[wife].marriageDate) { mDate = individuals[wife].marriageDate; mPlace = individuals[wife].marriagePlace; } if (!mDate && fam.sharedEventRef) { let evKey = '@' + fam.sharedEventRef + '@'; if (sharedEvents[evKey] && sharedEvents[evKey].date) { mDate = sharedEvents[evKey].date; mPlace = sharedEvents[evKey].place; } } let marriageInfo = { date: mDate, place: mPlace }; if (husb && wife) { if (!individuals[husb].spouses.includes(wife)) individuals[husb].spouses.push(wife); if (!individuals[wife].spouses.includes(husb)) individuals[wife].spouses.push(husb); if (!individuals[husb].marriages) individuals[husb].marriages = {}; if (!individuals[wife].marriages) individuals[wife].marriages = {}; individuals[husb].marriages[wife] = marriageInfo; individuals[wife].marriages[husb] = marriageInfo; } let parents = [husb, wife].filter(p => p !== null); kids.forEach(childId => { parents.forEach(parentId => { if (!individuals[childId].parents.includes(parentId)) individuals[childId].parents.push(parentId); if (!individuals[parentId].children.includes(childId)) individuals[parentId].children.push(childId); }); kids.forEach(siblingId => { if (siblingId !== childId && !individuals[childId].siblings.includes(siblingId)) { individuals[childId].siblings.push(siblingId); } }); }); }); let levels = {}; function assignLevels(personId, currentLevel, visited = new Set()) { if (visited.has(personId)) return; visited.add(personId); if (!(personId in levels) || currentLevel > levels[personId]) { levels[personId] = currentLevel; } let person = individuals[personId]; if (!person) return; person.spouses.forEach(spouseId => { if (!(spouseId in levels) || currentLevel > levels[spouseId]) { levels[spouseId] = currentLevel; } }); person.children.forEach(childId => { assignLevels(childId, currentLevel + 1, new Set(visited)); }); person.spouses.forEach(spouseId => { let spouse = individuals[spouseId]; if (spouse) { spouse.children.forEach(childId => { assignLevels(childId, currentLevel + 1, new Set(visited)); }); } }); } let roots = Object.values(individuals).filter(p => p.marriedName !== 'Neznámý' && p.parents.length === 0); if (roots.length === 0) roots = Object.values(individuals).filter(p => p.marriedName !== 'Neznámý'); roots.forEach(root => assignLevels(root.id, 0)); Object.values(individuals).forEach(p => { if (p.marriedName !== 'Neznámý' && !(p.id in levels)) levels[p.id] = 0; }); // PŘÍSNÝ FILTR NA PÁTEŘ: Osoba musí být v přímé sestupné linii (má rodiče v přímé linii A ZÁROVEŇ má dítě v přímé linii), NEBO to musí být koncový kořen / nejmladší potomek (Bohdan). // Tímto odfiltrujeme sourozence (jako je druhá Barbora), kteří sice mají rodiče v linii, ale sami nemají pokračovatele v přímém stromu. let spinePersons = Object.values(individuals).filter(p => { if (p.marriedName === 'Neznámý') return false; let isBohdan = p.children.length === 0 && p.parents.length > 0; // nebo nejmladší prvek let hasChildInTree = p.children.some(cId => individuals[cId] && individuals[cId].marriedName !== 'Neznámý'); let hasParentInTree = p.parents.some(pId => individuals[pId] && individuals[pId].marriedName !== 'Neznámý'); let isRoot = p.parents.length === 0 && p.children.length > 0; // Pro lidi uprostřed stromu (mezi kořenem a koncem) vyžadujeme, aby měli rodiče v linii A ZÁROVEŇ dítě v linii. let isMiddleSpine = hasParentInTree && hasChildInTree; return isRoot || isMiddleSpine || isBohdan; }); let genGroups = {}; spinePersons.forEach(p => { let lvl = levels[p.id] || 0; if (!genGroups[lvl]) genGroups[lvl] = []; genGroups[lvl].push(p); }); let nodes = []; let nodeMap = {}; let rowHeight = 145; let colWidth = 190; Object.keys(genGroups).sort((a,b) => a - b).forEach(lvl => { let personsInGen = genGroups[lvl]; let orderedPersons = []; let processed = new Set(); personsInGen.forEach(p => { if (processed.has(p.id)) return; orderedPersons.push(p); processed.add(p.id); let spouseId = p.spouses.find(sId => personsInGen.some(sp => sp.id === sId) && !processed.has(sId)); if (spouseId) { let spouse = individuals[spouseId]; let first = p; let second = spouse; if (p.sex === 'F' && spouse.sex === 'M') { first = spouse; second = p; } orderedPersons[orderedPersons.length - 1] = first; processed.add(first.id); orderedPersons.push(second); processed.add(second.id); } }); orderedPersons.forEach((p, index) => { let x = (index - orderedPersons.length / 2) * colWidth + colWidth / 2; let y = parseInt(lvl) * rowHeight + 60; let resolvedMedia = p.media.map(mId => mediaFiles[mId] ? mediaFiles[mId].file : mId); let resolvedSiblings = p.siblings.map(sId => individuals[sId]).filter(s => s); let nodeObj = { id: p.id, name: p.displayName, detailName: p.detailName, sex: p.sex, birth: p.birthDate, birthPlace: p.birthPlace, death: p.deathDate, deathPlace: p.deathPlace, occupation: p.occupation, marriages: p.marriages || {}, spouses: p.spouses, siblings: resolvedSiblings, hasSingleFamily: p.hasSingleFamily, media: resolvedMedia, x: x, y: y }; nodes.push(nodeObj); nodeMap[p.id] = nodeObj; }); }); let links = []; Object.values(families).forEach(fam => { let husb = fam.husb && nodeMap[fam.husb] ? fam.husb : null; let wife = fam.wife && nodeMap[fam.wife] ? fam.wife : null; let kids = fam.children.filter(c => c && nodeMap[c]); if (husb && wife) { links.push({ source: nodeMap[husb], target: nodeMap[wife], type: 'partner' }); } let parents = [husb, wife].filter(p => p !== null); parents.forEach(parent => { kids.forEach(child => { links.push({ source: nodeMap[parent], target: nodeMap[child], type: 'parent' }); }); }); }); document.getElementById('loading').style.display = 'none'; drawStructuredTree(nodes, links); }

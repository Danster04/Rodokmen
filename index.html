// Příklad úpravy funkce, která zpracovává osobu pro detail a pravou lištu function renderSidebarDetails(person, allPeople) { // 1. Základní údaje v pravé liště document.getElementById('det-jmeno').innerText = person.name; // ... tvoje ostatní vykreslování (daty, narození atd.) ... // 2. Vyhledání sourozenců, kteří NEJSOU v přímé linii stromu const sidebarSiblingsContainer = document.getElementById('det-sourozenci'); if (sidebarSiblingsContainer) { sidebarSiblingsContainer.innerHTML = ''; // Získání sourozenců (např. přes společné rodiče) let siblings = getSiblings(person, allPeople); if (siblings.length > 0) { let heading = document.createElement('h4'); heading.innerText = "Sourozenci (vedlejší linie):"; sidebarSiblingsContainer.appendChild(heading); let ul = document.createElement('ul'); siblings.forEach(sib => { let li = document.createElement('li'); let link = document.createElement('a'); link.href = "#"; link.innerText = sib.name; // Kliknutím na sourozence v pravé liště se přepne detail na něj link.onclick = (e) => { e.preventDefault(); renderSidebarDetails(sib, allPeople); }; li.appendChild(link); ul.appendChild(li); }); sidebarSiblingsContainer.appendChild(ul); sidebarSiblingsContainer.style.display = 'block'; } else { sidebarSiblingsContainer.style.display = 'none'; } } } // Pomocná funkce na nalezení sourozenců (pokud ji nemáš hotovou jinde) function getSiblings(person, allPeople) { if (!person.parents || person.parents.length === 0) return []; let siblingsSet = new Set(); person.parents.forEach(parentId => { let parent = allPeople.find(p => p.id === parentId); if (parent && parent.children) { parent.children.forEach(childId => { if (childId !== person.id) { let sib = allPeople.find(p => p.id === childId); if (sib) siblingsSet.add(sib); } }); } }); return Array.from(siblingsSet); }
